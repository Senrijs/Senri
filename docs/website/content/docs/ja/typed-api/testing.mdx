```ts twoslash title="app.test.ts"
import { createApp, defineRoutes } from "@senrijs/senri";
import { setCookie, getCookie } from "@senrijs/senri/cookie";
declare function expect(value: unknown): { toBe(expected: unknown): void };
const routes = defineRoutes({
    "POST /login": (_, c) => (setCookie(c, "session", "ok"), "ok"),
    "GET /me": (_, c) => ({ session: getCookie(c, "session") }),
});
const app = createApp();
app.routes(routes);
type Api = typeof routes;
const form = new FormData();
// ---cut---
import { request, testClient } from "@senrijs/senri/testing";

// 型付きクライアントで、サーバーを立てずに app.fetch を呼ぶ
const api = testClient<Api>(app, { cookies: true });
await api("POST /login");
const me = await api("GET /me");          // ログインの Cookie が付く
expect(me.status).toBe(200);

// 表にないルートや、型付きでは送れないリクエストは request() で(Response が返る)
const res = await request(app, "POST /upload", { body: form });
expect(res.status).toBe(404);
```

どのテストランナー(`bun test`・`node --test`・Vitest・`deno test`)でも使えます。

## testClient() と request()

| 関数 | 使うとき | 返すもの |
| --- | --- | --- |
| `testClient<Api>(app, options)` | 表のルートを、型付きで呼ぶ | [型付きクライアント](/docs/typed-api/client)と同じ結果 |
| `request(app, "METHOD /path", init)` | 表にないルート、型付きでは送れないリクエスト | `Response` {/* verify: request */} |

## オプション

- **接続元:** `127.0.0.1` として渡します。`getIp()`・`ipRestriction()`・開発モードの localhost の判定が、本物のサーバーと同じに動きます。`{ ip }` で変えられます {/* verify: ip */}
- **Cloudflare Workers のアプリ:** `{ env, ctx }` を渡します(`c.env`・`c.executionCtx`)。このときは Workers と同じく、接続元は渡しません {/* verify: workers */}
- **Cookie:** `{ cookies: true }` で、前の応答の `Set-Cookie` を、次のリクエストに付けます(名前ごとに 1 つ。domain と path は見ません) {/* verify: cookies */}
- **ヘッダー:** `{ headers }` で、毎回付けるヘッダーを決められます {/* verify: headers */}
